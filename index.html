iCalculate Total Salary of Each Employee
db.employees.aggregate([
  {
    $project: {
      name: 1, department: 1,
      total_salary: { $sum: ["$basic_salary", "$hra", "$allowances"] }
    }
  }
]);

iiFind Total Salary Paid by the Organization
db.employees.aggregate([
  {
    $group: {
      _id: "Total Salary",
